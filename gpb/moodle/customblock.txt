let elems=document.getElementsByClassName('kursbewertung');
if(elems.length>0) {
  let heute=new Date();
  heute=heute.getFullYear()+'-'+(heute.getMonth()+1<10 ? '0' : '')+(heute.getMonth()+1)+'-'+(heute.getDate()<10 ? '0' : '')+heute.getDate();
  let morgen=new Date();
  morgen.setDate(morgen.getDate()+1);
  morgen=morgen.getFullYear()+'-'+(morgen.getMonth()+1<10 ? '0' : '')+(morgen.getMonth()+1)+'-'+(morgen.getDate()<10 ? '0' : '')+morgen.getDate();
  let a=elems[elems.length-1];
  let url=a.href;
  let ab=null,bis=null;
  let i=url.indexOf('bewertungab=');
  if(i>=0) {
    i+='bewertungab='.length;
    ab=url.substring(i,i+10);
  }
  i=url.indexOf('bewertungbis=');
  if(i>=0) {
    i+='bewertungbis='.length;
    bis=url.substring(i,i+10);
  }
  if(ab && bis && morgen>=ab && heute<=bis) {
    let na=document.createElement('a');
    na.href=a.href;
    na.target=a.target;
    na.innerText='Kursbewertung';
    na.style='font-size:big;font-weight:bold;color:rgb(0,128,0);border:1px solid rgb(0,128,0);padding:4px;'
    document.getElementById('kursbewertungreminder').appendChild(na);
  }
}
</script>
